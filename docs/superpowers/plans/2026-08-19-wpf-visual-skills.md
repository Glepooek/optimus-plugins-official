# WPF 视觉还原 Skill 组（组件库 + 页面组装）Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在 `optimus-frontend-plugin` 中落地"MasterGo 设计稿 → 可复用组件库 → 组件组装页面"链路：新增 `mastergo-to-wpf-components` skill，升级 `mastergo-to-wpf` 为组件驱动组装，并新增 `wpf-project-conventions` 共享约定。

**Architecture:** 三个产物围绕一个纯函数核心：`extract_components.py`（DSL JSON → 组件索引 + 颜色/DataTemplate 资源）与 `match_components.py`（设计稿节点 → 组件库命中/缺失报告）均为离线可测脚本；SKILL.md 负责 MCP 读取、确认门、冲突决策与编译验证；`wpf-project-conventions/CONVENTIONS.md` 是项目工程约定的事实来源，两个 skill 必须先读。

**Tech Stack:** Python 3（unittest，仓库本机无 pytest）、MasterGo Magic MCP（仓库内置）、WPF/XAML、.NET SDK（`dotnet build` 验证）、JSON。

```python
"""Contract tests for extract_components.py (offline, unittest only)."""

import json
import subprocess
import sys
import tempfile
import unittest
from pathlib import Path

ASSETS = Path(__file__).parent / "assets"
SCRIPT = Path(__file__).parent / "extract_components.py"


def run_cli(asset: str, mapping: str | None = None) -> tuple[subprocess.CompletedProcess, Path]:
    with tempfile.TemporaryDirectory() as tmp:
        src = Path(tmp) / "input"
        out = Path(tmp) / "out"
        src.mkdir()
        payload = json.loads((ASSETS / asset).read_text(encoding="utf-8"))
        listing = payload.get("list", {})
        (src / "sections-list.json").write_text(
            json.dumps({"rootMetadata": listing.get("rootMetadata", {})}, ensure_ascii=False), encoding="utf-8"
        )
        for i, section in enumerate(payload["sections"]):
            (src / f"section-{i}.json").write_text(json.dumps(section, ensure_ascii=False), encoding="utf-8")
        if mapping:
            mapping_path = Path(tmp) / "mapping.json"
            mapping_path.write_text(mapping, encoding="utf-8")
            return subprocess.run(
                [sys.executable, str(SCRIPT), "--input", str(src), "--out", str(out), "--mapping", str(mapping_path)],
                capture_output=True, text=True,
            ), out
        return subprocess.run(
            [sys.executable, str(SCRIPT), "--input", str(src), "--out", str(out)],
            capture_output=True, text=True,
        ), out


class ExtractCliTestCase(unittest.TestCase):
    def test_instance_repeated_extracts_style(self):
        proc, out = run_cli("instance-repeated.json")
        self.assertEqual(proc.returncode, 0, proc.stderr)
        index = json.loads((out / "components-index.json").read_text(encoding="utf-8"))
        self.assertEqual(len(index["components"]), 1)
        item = index["components"][0]
        self.assertEqual(item["kind"], "style")
        self.assertEqual(item["status"], "new")
        self.assertEqual(item["occurrences"], 2)

    def test_single_instance_not_extracted(self):
        proc, out = run_cli("single-instance.json")
        self.assertEqual(proc.returncode, 0, proc.stderr)
        index = json.loads((out / "components-index.json").read_text(encoding="utf-8"))
        self.assertEqual(index["components"], [])

    def test_repeated_frames_extract_datatemplate(self):
        proc, out = run_cli("repeated-frames.json")
        self.assertEqual(proc.returncode, 0, proc.stderr)
        index = json.loads((out / "components-index.json").read_text(encoding="utf-8"))
        self.assertEqual(index["components"][0]["kind"], "datatemplate")
        self.assertIn("FRAME.Card", (out / "DataTemplates.generated.xaml").read_text(encoding="utf-8"))
        self.assertIn("FillFill2", (out / "Colors.generated.xaml").read_text(encoding="utf-8"))

    def test_mapped_instance_uses_mapping(self):
        mapping = json.dumps({
            "resources": {},
            "xmlns": {"c": "clr-namespace:Optimus.Controls;assembly=Optimus.Desktop"},
            "components": {
                "ActionButton.Primary": {
                    "control": "c:ActionButton",
                    "variants": {"Default": ["Background", "CornerRadius"]},
                }
            },
        }, ensure_ascii=False)
        proc, out = run_cli("mapped-instance.json", mapping)
        self.assertEqual(proc.returncode, 0, proc.stderr)
        index = json.loads((out / "components-index.json").read_text(encoding="utf-8"))
        item = index["components"][0]
        self.assertEqual(item["status"], "mapped")
        self.assertEqual(item["kind"], "control")

    def test_broken_token_hard_stops(self):
        proc, _ = run_cli("broken-token.json")
        self.assertEqual(proc.returncode, 2)
        self.assertTrue(proc.stderr.startswith("error: "), proc.stderr)


if __name__ == "__main__":
    unittest.main()
```

创建 `plugins/optimus-frontend-plugin/skills/mastergo-to-wpf-components/scripts/extract_components.py`：

```python
"""Extract reusable WPF visual components from MasterGo DSL JSON.

Pure function: no network, no MCP, no environment access beyond CLI args.
Reads section-*.json plus sections-list.json from --input, writes
components-index.json, Colors.generated.xaml and DataTemplates.generated.xaml
to --out.

Exit contract:
- 0: success; JSON report printed to stdout
- 2: hard error; single-line message on stderr, no stdout
"""

from __future__ import annotations

import argparse
import json
import re
import sys
from collections import Counter
from pathlib import Path

HEX_COLOUR = re.compile(r"^#([0-9A-Fa-f]{6}|[0-9A-Fa-f]{8})$")
KEY_SUFFIX = re.compile(r"[/ -]+")


class ExtractionError(Exception):
    """Hard error that terminates with exit 2."""


def read_json(path: Path) -> object:
    try:
        return json.loads(path.read_text(encoding="utf-8"))
    except (OSError, json.JSONDecodeError) as exc:
        raise ExtractionError(f"cannot read {path}: {exc}") from exc


def load_sections(directory: Path) -> list[dict]:
    listing_path = directory / "sections-list.json"
    if listing_path.exists():
        listing = read_json(listing_path)
        if not isinstance(listing, dict):
            raise ExtractionError("sections-list.json must be a JSON object")
        if not isinstance(listing.get("rootMetadata"), dict):
            raise ExtractionError("sections-list.json missing rootMetadata")
    sections: list[dict] = []
    for path in sorted(
        directory.glob("section-*.json"), key=lambda p: int(p.stem.split("-")[1])
    ):
        payload = read_json(path)
        if not isinstance(payload, dict):
            raise ExtractionError(f"{path.name} must be a JSON object")
        if not isinstance(payload.get("nodes"), list):
            raise ExtractionError(f"{path.name} missing nodes array")
        sections.append(payload)
    if not sections:
        raise ExtractionError("no section-*.json files found")
    return sections


def walk(nodes: list) -> list[dict]:
    result: list[dict] = []
    for node in nodes:
        if not isinstance(node, dict):
            continue
        result.append(node)
        children = node.get("children")
        if isinstance(children, list):
            result.extend(walk(children))
    return result


def resource_key(token: str) -> str:
    """'Text/Text-4' -> 'TextText4' (matches existing dsl_to_xaml.py)."""
    return KEY_SUFFIX.sub("", token)


def node_colour(node: dict, styles: dict) -> str | None:
    token = node.get("_token")
    if isinstance(token, str) and token:
        return None  # token colours are resolved separately
    colour = node.get("_color")
    if isinstance(colour, str) and HEX_COLOUR.match(colour.strip()):
        return colour.strip()
    reference = node.get("fill")
    if isinstance(reference, str) and reference:
        entry = styles.get(reference)
        if not isinstance(entry, dict):
            raise ExtractionError(f"fill {reference!r} not in dsl.styles")
        values = entry.get("value")
        if isinstance(values, list) and values and isinstance(values[0], dict):
            c = values[0].get("color")
            if isinstance(c, str) and HEX_COLOUR.match(c.strip()):
                return c.strip()
    return None


def frame_signature(node: dict) -> tuple:
    return (
        node.get("type"),
        node.get("name"),
        node.get("_token"),
        node.get("_color"),
        len(node.get("children") or []),
    )


def load_mapping(path_value: str | None) -> dict:
    if not path_value:
        return {}
    mapping = read_json(Path(path_value))
    if not isinstance(mapping, dict):
        raise ExtractionError("mapping must be a JSON object")
    components = mapping.get("components")
    return components if isinstance(components, dict) else {}


def build_component_index(sections: list[dict], mapping: dict) -> list[dict]:
    nodes_with_styles: list[tuple[dict, dict]] = []
    for section in sections:
        styles = section.get("styles")
        if not isinstance(styles, dict):
            styles = {}
        for node in walk(section.get("nodes") or []):
            nodes_with_styles.append((node, styles))
    instance_names = Counter(
        node.get("name") for node, _ in nodes_with_styles if node.get("type") == "INSTANCE"
    )
    frame_sigs = Counter(
        frame_signature(node) for node, _ in nodes_with_styles if node.get("type") in ("FRAME", "GROUP")
    )

    index: list[dict] = []
    seen: set[str] = set()
    for node, _ in nodes_with_styles:
        if node.get("type") == "INSTANCE":
            name = node.get("name")
            if not isinstance(name, str) or not name or name in seen:
                continue
            seen.add(name)
            if name in mapping:
                definition = mapping[name]
                index.append({
                    "key": name,
                    "kind": "control" if isinstance(definition, dict) and definition.get("control") else "style",
                    "source": [name],
                    "occurrences": instance_names[name],
                    "status": "mapped",
                    "resourceKey": name,
                })
                node_colour(node, styles)  # validate fill references of mapped instances
            elif instance_names[name] >= 2:
                index.append({
                    "key": name,
                    "kind": "style",
                    "source": [name],
                    "occurrences": instance_names[name],
                    "status": "new",
                    "resourceKey": name,
                })
                node_colour(node, styles)  # validate fill references of repeated instances
        elif node.get("type") in ("FRAME", "GROUP"):
            sig = frame_signature(node)
            if frame_sigs[sig] < 2:
                continue
            key = f"{node.get('type')}.{node.get('name')}"
            if key in seen:
                continue
            seen.add(key)
            index.append({
                "key": key,
                "kind": "datatemplate",
                "source": [node.get("name")],
                "occurrences": frame_sigs[sig],
                "status": "new",
                "resourceKey": key,
            })
            node_colour(node, styles)  # validate fill references of extracted frames
    return index


def collect_tokens(sections: list[dict], index: list[dict]) -> dict[str, str]:
    keys = {item["resourceKey"] for item in index}
    tokens: dict[str, str] = {}
    for section in sections:
        styles = section.get("styles")
        if not isinstance(styles, dict):
            styles = {}
        for node in walk(section.get("nodes") or []):
            node_key = (
                node.get("name")
                if node.get("type") == "INSTANCE"
                else f"{node.get('type')}.{node.get('name')}"
            )
            if node_key not in keys:
                continue
            token = node.get("_token")
            if isinstance(token, str) and token:
                tokens.setdefault(resource_key(token), node_colour(node, styles) or "#000000")
    return tokens


def xaml_escape(value: str) -> str:
    return value.replace("&", "&amp;").replace("<", "&lt;").replace(">", "&gt;")


def render_colors_xaml(tokens: dict[str, str]) -> str:
    lines = [
        '<ResourceDictionary xmlns="http://schemas.microsoft.com/winfx/2006/xaml/presentation"',
        '                    xmlns:x="http://schemas.microsoft.com/winfx/2006/xaml">',
    ]
    for key, colour in sorted(tokens.items()):
        lines.append(f'    <SolidColorBrush x:Key="{key}" Color="{colour}"/>')
    lines.append("</ResourceDictionary>")
    return "\n".join(lines) + "\n"


def render_data_templates_xaml(sections: list[dict], index: list[dict]) -> str:
    templates = {item["resourceKey"]: item for item in index if item["kind"] == "datatemplate"}
    lines = [
        '<ResourceDictionary xmlns="http://schemas.microsoft.com/winfx/2006/xaml/presentation"',
        '                    xmlns:x="http://schemas.microsoft.com/winfx/2006/xaml">',
    ]
    emitted: set[str] = set()
    for section in sections:
        for node in walk(section.get("nodes") or []):
            key = f"{node.get('type')}.{node.get('name')}"
            if key not in templates or key in emitted:
                continue
            emitted.add(key)
            labels = [
                n.get("name")
                for n in (node.get("children") or [])
                if isinstance(n, dict) and n.get("type") == "TEXT"
            ]
            lines.append(f'    <DataTemplate x:Key="{key}">')
            lines.append("        <Grid>")
            for label in labels[:6]:
                lines.append(f'            <TextBlock Text="{xaml_escape(str(label))}"/>')
            lines.append("        </Grid>")
            lines.append("    </DataTemplate>")
    lines.append("</ResourceDictionary>")
    return "\n".join(lines) + "\n"


def main(argv: list[str]) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--input", required=True, help="directory with sections-list.json + section-*.json")
    parser.add_argument("--out", required=True, help="output directory for generated component library")
    parser.add_argument("--mapping", help="optional strict wpf-project-mapping.json path")
    args = parser.parse_args(argv)
    try:
        in_dir = Path(args.input)
        out_dir = Path(args.out)
        if not in_dir.is_dir():
            raise ExtractionError(f"input directory not found: {in_dir}")
        sections = load_sections(in_dir)
        mapping = load_mapping(args.mapping)
        index = build_component_index(sections, mapping)
        tokens = collect_tokens(sections, index)
        out_dir.mkdir(parents=True, exist_ok=True)
        (out_dir / "components-index.json").write_text(
            json.dumps({"version": 1, "components": index}, ensure_ascii=False, indent=2),
            encoding="utf-8",
        )
        (out_dir / "Colors.generated.xaml").write_text(render_colors_xaml(tokens), encoding="utf-8")
        (out_dir / "DataTemplates.generated.xaml").write_text(
            render_data_templates_xaml(sections, index), encoding="utf-8"
        )
        print(json.dumps({"extracted": len(index), "tokens": len(tokens)}, ensure_ascii=False))
        return 0
    except ExtractionError as exc:
        print(f"error: {exc}", file=sys.stderr)
        return 2


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))
```

```python
        payload = json.loads((ASSETS / asset).read_text(encoding="utf-8"))
```

```python
"""Match MasterGo DSL nodes against a WPF component library index.

Pure function: no network, no MCP. Reads section-*.json (--input) and
components-index.json (--index), writes component-match-report.json to --out.

Exit contract:
- 0: success; report written, JSON summary on stdout
- 2: hard error; single-line message on stderr, no stdout
"""

from __future__ import annotations

import argparse
import json
import sys
from pathlib import Path


class MatchError(Exception):
    """Hard error that terminates with exit 2."""


def read_json(path: Path) -> object:
    try:
        return json.loads(path.read_text(encoding="utf-8"))
    except (OSError, json.JSONDecodeError) as exc:
        raise MatchError(f"cannot read {path}: {exc}") from exc


def load_index(path: Path) -> dict[str, str]:
    payload = read_json(path)
    if not isinstance(payload, dict):
        raise MatchError("components-index.json must be a JSON object")
    components = payload.get("components")
    if not isinstance(components, list):
        raise MatchError("components-index.json components must be an array")
    keys: dict[str, str] = {}
    for item in components:
        if not isinstance(item, dict):
            raise MatchError("components-index.json entry must be an object")
        name = item.get("resourceKey")
        if isinstance(name, str) and name:
            keys[name] = item.get("kind", "style")
    return keys


def walk(nodes: list) -> list[dict]:
    result: list[dict] = []
    for node in nodes:
        if not isinstance(node, dict):
            continue
        result.append(node)
        children = node.get("children")
        if isinstance(children, list):
            result.extend(walk(children))
    return result


def match(sections: list[dict], index_keys: dict[str, str]) -> tuple[list[dict], list[dict]]:
    matches: list[dict] = []
    missing: list[dict] = []
    for section in sections:
        for node in walk(section.get("nodes") or []):
            node_type = node.get("type")
            name = node.get("name")
            if not isinstance(name, str) or not name:
                continue
            if node_type == "INSTANCE":
                candidate = name
            elif node_type in ("FRAME", "GROUP"):
                candidate = f"{node_type}.{name}"
            else:
                continue
            if candidate in index_keys:
                matches.append({
                    "nodeId": node.get("id"),
                    "nodeName": name,
                    "resourceKey": candidate,
                    "kind": index_keys[candidate],
                })
            elif node_type == "INSTANCE":
                missing.append({
                    "nodeId": node.get("id"),
                    "nodeName": name,
                    "reason": "未在组件库注册的实例",
                })
    return matches, missing


def main(argv: list[str]) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--input", required=True, help="directory with sections-list.json + section-*.json")
    parser.add_argument("--index", required=True, help="components-index.json path")
    parser.add_argument("--out", required=True, help="output directory for the match report")
    args = parser.parse_args(argv)
    try:
        in_dir = Path(args.input)
        if not in_dir.is_dir():
            raise MatchError(f"input directory not found: {in_dir}")
        sections: list[dict] = []
        for path in sorted(in_dir.glob("section-*.json"), key=lambda p: int(p.stem.split("-")[1])):
            payload = read_json(path)
            if not isinstance(payload, dict) or not isinstance(payload.get("nodes"), list):
                raise MatchError(f"{path.name} missing nodes array")
            sections.append(payload)
        if not sections:
            raise MatchError("no section-*.json files found")
        index_keys = load_index(Path(args.index))
        matches, missing = match(sections, index_keys)
        out_dir = Path(args.out)
        out_dir.mkdir(parents=True, exist_ok=True)
        (out_dir / "component-match-report.json").write_text(
            json.dumps({"matches": matches, "missing": missing}, ensure_ascii=False, indent=2),
            encoding="utf-8",
        )
        print(json.dumps({"matched": len(matches), "missing": len(missing)}, ensure_ascii=False))
        return 0
    except MatchError as exc:
        print(f"error: {exc}", file=sys.stderr)
        return 2


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))
```
